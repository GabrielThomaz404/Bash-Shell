#!/bin/bash

echo "CALCULADORA"

echo "Digite o primeiro número:"
read num1

echo "Digite o segundo número:"
read num2

echo "1 - Soma"
echo "2 - Subtração"
echo "3 - Multiplicação"
echo "4 - Divisão"
echo "5 - Sair"


echo "Escolha uma opção:"
read opcao

if [ "$opcao" -eq 1 ]; then
    resultado=$(expr $num1 + $num2)
    echo "Resultado da soma: $resultado"

elif [ "$opcao" -eq 2 ]; then
    resultado=$(expr $num1 - $num2)
    echo "Resultado da subtração: $resultado"

elif [ "$opcao" -eq 3 ]; then
    resultado=$(expr $num1 \* $num2)
    echo "Resultado da multiplicação: $resultado"

elif [ "$opcao" -eq 4 ]; then
    if [ "$num2" -eq 0 ]; then
        echo "Erro: não é possível dividir por zero."
    else
        resultado=$(expr $num1 / $num2)
        echo "Resultado da divisão: $resultado"
    fi

elif [ "$opcao" -eq 5 ]; then
    echo "Saindo da calculadora..."

else
    echo "Opção inválida! Escolha uma opção de 1 a 5."
fi

#!/bin/bash

echo "ENCERRAR PROCESSO"

echo "Digite o PID do processo que deseja encerrar:"
read pid

if ps -ef | grep -w "$pid" | grep -v grep > /dev/null; then

    echo ""
    echo "PID informado: $pid"
    echo "O processo foi encontrado."
    echo "Deseja realmente encerrar esse processo? (s/n)"
    read resposta

    if [ "$resposta" = "s" ]; then

        kill "$pid"
        echo "Processo $pid encerrado."

    else

        echo "Operação cancelada."

    fi

else

    echo "PID $pid não foi encontrado entre os processos em execução."

fi
